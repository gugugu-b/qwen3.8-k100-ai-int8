```bash
  --model-path /data/model/Qwen3.8-27B-Channel-INT8-w8a8 \
  --tp-size 2 --pp-size 1 \
  --attention-backend fa3 --mm-attention-backend fa3 \

  --speculative-eagle-topk 1 --speculative-num-draft-tokens 2 \
  --custom-all-reduce-backend native
```
